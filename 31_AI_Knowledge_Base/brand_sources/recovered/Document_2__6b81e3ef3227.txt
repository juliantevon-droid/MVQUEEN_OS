@dataclass
class PlatformConfig:
    """
    Centralized platform configuration.
    This is what the Control Panel reads and writes.
    """

    editorial_mode: str = "standard"
    overwrite_mode: str = "balanced"
    persona_intensity: str = "balanced"
    seo_mode: str = "standard"
    metafield_mode: str = "standard"

    # Android-safe paths
    inputcsvpath: str = "/storage/emulated/0/mvqueen/input.csv"
    outputcsvpath: str = "/storage/emulated/0/mvqueen/output.csv"

    # Debug
    debug: bool = True

    # Extra settings
    extra: Dict[str, Any] = field(default_factory=dict)

    def geteditorialsettings(self):
        return EDITORIALMODES.get(self.editorialmode, EDITORIAL_MODES["standard"])

    def getoverwritesettings(self):
        return OVERWRITEMODES.get(self.overwritemode, OVERWRITE_MODES["balanced"])

    def getpersonastrength(self):
        return PERSONAINTENSITY.get(self.personaintensity, 2)

    def getseosettings(self):
        return SEOMODES.get(self.seomode, SEO_MODES["standard"])

    def getmetafieldsettings(self):
        return METAFIELDMODES.get(self.metafieldmode, METAFIELD_MODES["standard"])

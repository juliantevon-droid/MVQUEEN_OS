From mvqueen_engine.config import SHOPIFY_STORE_DOMAIN, SHOPIFY_API_VERSION

Def fetch_all_products():
    “””Fetch all Shopify products and return handle → product_id map.”””
    Products = {}
    url = fhttps://{SHOPIFY_STORE_DOMAIN}/admin/api/{SHOPIFY_API_VERSION}/products.json?limit=250

    while url:
        r = requests.get(url, headers=get_headers())
        r.raise_for_status()

        data = r.json().get(“products”, [])
        for p in data:
            products[p[“handle”]] = p[“id”]

        url = r.links.get(“next”, {}).get(“url”)

    return products

Product Updater — MVQueen Engine
Handles safe product updates, metafields, and DRY RUN mode.
“””

Import time
Import requests
From mvqueen_engine.shopify_api.auth import get_headers
From mvqueen_engine.shopify_api.session_manager import fetch_all_products
From mvqueen_engine.config import SHOPIFY_STORE_DOMAIN, SHOPIFY_API_VERSION

Def update_products(df, dry_run=True, sleep_seconds=0.4):
    “””Update Shopify products safely.”””
    Product_map = fetch_all_products()

    For idx, row in df.iterrows():
        Handle = row[“Handle”]

        If handle not in product_map:
            Print(f”Skipping missing product: {handle}”)
            Continue

        Product_id = product_map[handle]

        Payload = {
            “product”: {
                “id”: product_id,
                “title”: row[“Title”],
                “body_html”: row[“Body (HTML)”],
                “tags”: row[“Tags”],
                “metafields_global_title_tag”: row[“SEO Title”],
                “metafields_global_description_tag”: row[“SEO Description”],
            }
        }

        # Build metafields
        Metafields = []
        For col in row.index:
            If col.startswith(“metafield.custom.”):
                Value = str(row[col]).strip()
                If value:
                    Key = col.replace(“metafield.custom.”, “”)
                    Metafields.append({
                        “namespace”: “custom”,
                        “key”: key,
                        “type”: “single_line_text_field”,
                        “value”: value
                    })

        If metafields:
            Payload[“product”][“metafields”] = metafields

        If dry_run:
            Print(f”DRY RUN → Would update {handle}”)
            Continue

        Try:
            R = requests.put(
                Fhttps://{SHOPIFY_STORE_DOMAIN}/admin/api/{SHOPIFY_API_VERSION}/products/{product_id}.json,
                Headers=get_headers(),
                Json=payload
            )
            r.raise_for_status()
            print(f”Updated {handle}”)
        except Exception as e:
            print(f”Failed {handle}: {e}”)

        time.sleep(sleep_seconds)

Def generate_metafields(handle: str):
    “””Generate a full metafield dictionary for a product.”””

        “metafield.custom.texture”: deterministic_choice(TEXTURES, handle, 3),
        “metafield.custom.finish”: deterministic_choice(FINISHES, handle, 4),
        “metafield.custom.mood”: deterministic_choice(MOODS, handle, 5),
        “metafield.custom.occasion”: deterministic_choice(OCCASIONS, handle, 6),      “metafield.custom.business_tier”: deterministic_choice(BUSINESS_TIERS, handle, 7),

        # SEO metafields
        “metafield.custom.search_priority”: “true”,        “metafield.custom.editorial_ready”: “true”,
    }

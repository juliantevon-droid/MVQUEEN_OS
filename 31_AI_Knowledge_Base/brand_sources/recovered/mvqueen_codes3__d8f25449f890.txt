    """Load CSV, clean text, generate metafields, return optimized DataFrame."""

    # Clean HTML + whitespace
    if "Body (HTML)" in df.columns:
        df["Body (HTML)"] = df["Body (HTML)"].astype(str).apply(strip_html).apply(normalize_whitespace).apply(enforce_brand)

    # Generate metafields

Generates tags based on category, product type, benefits, persona, fashion vocab, SEO, and vibes.

From mvqueen_engine.brand_brain.brand_banks import (
    TAG_RULES, FASHION_VIBES, FASHION_MATERIALS, SEO_PRIMARY, SEO_SECONDARY
)
From mvqueen_engine.utils.deterministic import deterministic_choice, deterministic_multi

Def generate_tags(row):
    Tags = set()

    # Category tags
    Category = row.get(“Category”, “”)
    If category in TAG_RULES:
        Tags.update(TAG_RULES[category])

    # Product type tag
    Ptype = row.get(“Product Type”, “”)
    If ptype:
        Tags.add(ptype.lower())

    # Benefit tags
    Benefits = row.get(“metafield.custom.key_benefits”, “”)
    For b in benefits.split(“,”):
        Tags.add(b.strip().lower())

    # Fashion vibes
    Vibe = deterministic_choice(FASHION_VIBES, row[“Handle”], 0)
    Tags.add(vibe.replace(“ “, “-“))

    # Materials (if apparel)
    If category == “Apparel”:
        Material = deterministic_choice(FASHION_MATERIALS, row[“Handle”], 1)
        Tags.add(material.replace(“ “, “-“))

    # SEO tags
    Primary = deterministic_choice(SEO_PRIMARY, row[“Handle”], 2)
    Secondary = deterministic_choice(SEO_SECONDARY, row[“Handle”], 3)
    Tags.add(primary.replace(“ “, “-“))
    Tags.add(secondary.replace(“ “, “-“))

    Return “, “.join(sorted(tags))

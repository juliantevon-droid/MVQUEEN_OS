MVQueen OS - Module 4: Brand Voice Linter Script
This document contains the complete Google Apps Script code for the MVQueen OS Brand Voice Linter. It scans markdown files across all 40+ folders for forbidden low-end copy or improper styling rules, logging violations directly inside a third tab titled "Brand Voice Audits" in your centralized spreadsheet.
1. The Apps Script Code
/** * MVQueen OS - Brand Voice Linter * Deep-scans text/markdown contents to ensure absolute luxury copy compliance. */function lintMVQUEENBrandVoice() {  const ROOT_FOLDER_ID = 'YOUR_GOOGLE_DRIVE_FOLDER_ID_HERE';   const rootFolder = DriveApp.getFolderById(ROOT_FOLDER_ID);  const subFolders = rootFolder.getFolders();    // Banned low-end buzzwords or structural style rules  const FORBIDDEN_WORDS = [/girlboss/i, /hustle/i, /cheap/i, /bargain/i, /budget-friendly/i];  const EXCLAMATION_LIMIT = 2; // Maximum allowed exclamation marks per file to preserve luxury tone    let violations = [];  Logger.log("Commencing MVQueen OS luxury brand voice alignment sweep...");    while (subFolders.hasNext()) {    let folder = subFolders.next();    let folderName = folder.getName();    let files = folder.getFiles();        while (files.hasNext()) {      let file = files.next();      let fileName = file.getName();      let mime = file.getMimeType();            if (mime === MimeType.PLAIN_TEXT || fileName.endsWith('.md')) {        let content = file.getAs('text/plain').getDataAsString();        let issuesFound = [];                // 1. Evaluate banned vocabulary matches        FORBIDDEN_WORDS.forEach(regex => {          if (regex.test(content)) {            issuesFound.push(`Banned vocabulary match: "${content.match(regex)[0]}"`);          }        });                // 2. Evaluate structural styling constraints (Exclamation points check)        let exclamationCount = (content.match(/!/g) || []).length;        if (exclamationCount > EXCLAMATION_LIMIT) {          issuesFound.push(`Excessive punctuation: ${exclamationCount} exclamation marks used`);        }                // Log to system array if parameters are broken        if (issuesFound.length > 0) {          violations.push([            fileName,            folderName,            issuesFound.join(' | '),            new Date().toLocaleDateString()          ]);        }      }    }  }    // Output logs directly to a third tracking sheet tab  let reportFiles = rootFolder.getFilesByName("MVQUEEN_OS_Health_Report");  if (reportFiles.hasNext()) {    let ss = SpreadsheetApp.openById(reportFiles.next().getId());    let lintSheet = ss.getSheetByName("Brand Voice Audits");        if (!lintSheet) {      lintSheet = ss.insertSheet("Brand Voice Audits");      lintSheet.appendRow(["Asset File Name", "Location Folder", "Stylistic Violations Detected", "Last Scanned"]);      lintSheet.getRange("A1:D1").setFontWeight("bold");    } else {      if (lintSheet.getLastRow() > 1) {        lintSheet.deleteRows(2, lintSheet.getLastRow() - 1);      }    }        if (violations.length > 0) {      lintSheet.getRange(2, 1, violations.length, 4).setValues(violations);      Logger.log("Brand Voice Linter complete. Deviations flagged.");    } else {      lintSheet.appendRow(["All assets perfectly aligned with luxury brand standards.", "N/A", "None", new Date().toLocaleDateString()]);      Logger.log("Brand Voice Linter complete. 100% brand voice alignment verified.");    }  }}
